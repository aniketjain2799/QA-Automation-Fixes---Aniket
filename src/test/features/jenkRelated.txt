const path = require("path");
const reporter = require("cucumber-html-reporter");
const fs = require("fs");

const currentDateTime = new Date().toLocaleString();
const testEnv = process.env.ENV || "STAGING";

const jsonFileName = process.argv[2] || "cucumber-report.json";
const htmlFileName = process.argv[3] || "cucumber-report.html";

const jsonFilePath = path.join(process.cwd(), "test-results", jsonFileName);
const htmlFilePath = path.join(process.cwd(), "test-results", htmlFileName);

// Check if JSON report exists before proceeding
if (!fs.existsSync(jsonFilePath)) {
  console.error("Cucumber JSON report not found:", jsonFilePath);
  process.exit(1);
}

const options = {
  theme: "bootstrap",
  jsonFile: jsonFilePath,
  output: htmlFilePath,
  reportSuiteAsScenarios: true,
  scenarioTimestamp: true,
  launchReport: true,
  metadata: {
    Project: "Readibots",
    "App Version": "1.0.0",
    Cycle: "Regression",
    Browser: "Chrome",
    "Test Environment": testEnv,
    Parallel: "Scenarios",
    Platform: "Ubuntu 24LTS",
    Executed: "GitHub Cloud",
    "Report Generated": currentDateTime,
  },
  failedSummaryReport: false,
  brandTitle: "Readibots Test Automation Report",
  name: "Automation Results",
};

let reportData;
try {
  reportData = JSON.parse(fs.readFileSync(jsonFilePath, "utf-8"));
} catch (error) {
  console.error("Error reading Cucumber JSON report:", error);
  process.exit(1);
}

let totalScenarios = 0;
let passedScenarios = 0;
let failedScenarios = 0;

if (Array.isArray(reportData)) {
  reportData.forEach((feature) => {
    if (Array.isArray(feature.elements)) { // Ensure it's an array before iterating
      feature.elements.forEach((scenario) => {
        totalScenarios++;
        const isFailed = scenario.steps.some(
          (step) => step.result && step.result.status === "failed"
        );
        if (isFailed) {
          failedScenarios++;
        } else {
          passedScenarios++;
        }
      });
    }
  });
}

// Ensure scenario_counts.txt is saved in Jenkins workspace
const scenarioCountsPath = path.join(process.cwd(), "scenario_counts.txt");
const scenarioCountsContent = `Total Scenarios: ${totalScenarios}\nPassed Scenarios: ${passedScenarios}\nFailed Scenarios: ${failedScenarios}\n`;

fs.writeFileSync(scenarioCountsPath, scenarioCountsContent);
console.log(`Scenario counts saved to: ${scenarioCountsPath}`);
console.log(`Total: ${totalScenarios}, Passed: ${passedScenarios}, Failed: ${failedScenarios}`);

reporter.generate(options);

import { BeforeAll, AfterAll, Before, After, Status, setDefaultTimeout } from "@cucumber/cucumber";
import { Browser, BrowserContext } from "@playwright/test";
import { fixture } from "./pageFixture";
import { invokeBrowser } from "../helper/browsers/browserManager";
import { getEnv } from "../helper/env/env";
import { createLogger } from "winston";
import { options } from "../helper/util/logger";
import * as fs from "fs-extra";

let browser: Browser;
let context: BrowserContext;
let scenarioStartTime: number;
let testResults: any[] = [];

setDefaultTimeout(1200 * 1000);

BeforeAll(async function () {
    getEnv();
    browser = await invokeBrowser();
});

Before({ tags: "not @auth" }, async function ({ pickle }) {
    scenarioStartTime = performance.now();
    const scenarioName = pickle.name + pickle.id;
    context = await browser.newContext({
        permissions: ["clipboard-read"],
        recordVideo: {
            dir: "test-results/videos",
        },
    });
    fixture.context = context;
    await context.tracing.start({
        name: scenarioName,
        title: pickle.name,
        sources: true,
        screenshots: true,
        snapshots: true
    });
    const page = await context.newPage();
    await page.setViewportSize({ width: 1355, height: 629 });
    fixture.page = page;
    fixture.logger = createLogger(options(scenarioName));
});

After(async function ({ pickle, result }) {
    let img: Buffer;
    const traceFilePath = `./test-results/trace/${pickle.id}.zip`;

    const scenarioEndTime = performance.now();
    const scenarioExecutionTime = scenarioEndTime - scenarioStartTime;

    const executionMinutes = Math.floor(scenarioExecutionTime / 60000);
    const executionSeconds = Math.floor((scenarioExecutionTime % 60000) / 1000);
    const formattedExecutionTime = `${executionMinutes}m ${executionSeconds.toString().padStart(2, "0")}s`;

    const filePath = "./test-results/testResults.json";

    if (fs.existsSync(filePath)) {
        testResults = fs.readJsonSync(filePath);
    }

    const testResult = {
        testId: testResults.length + 1,
        name: pickle.name,
        status: result?.status === Status.PASSED ? "OK" : "FAIL",
        verified: new Date().toISOString(),
        executionTime: formattedExecutionTime,
        subTests: [
            "Login to Bot Studio",
            "Create Environment",
            "Expand Environment",
            "Create New Folder",
            "Create Bot Queue and Data Table",
            "Send a Script in Data Table",
            "Run Data Table",
            "Run Service Bot",
            "Run Bot",
            "Send Data from Service Bot to Data Table",
            "Validate Total Data Count"
        ]
    };

    testResults.push(testResult);
    fs.writeJsonSync(filePath, testResults, { spaces: 2 });

    if (result?.status == Status.FAILED) {
        img = await fixture.page.screenshot({
            path: `./test-results/screenshots/${pickle.name}.png`, type: "png"
        });
        await this.attach(img, "image/png");
    }

    await context.tracing.stop({ path: traceFilePath });
    await fixture.page.close();
    await context.close();
});



AfterAll(async function () {
    await browser.close();

    try {
        const filePath = "./test-results/testResults.json";
        let testResults = [];

        if (fs.existsSync(filePath)) {
            testResults = fs.readJsonSync(filePath);
        }

        if (testResults.length === 0) {

        const payload = {

            environment: process.env.ENV || "STAGING",

        const response = await fetch(apiUrl, {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
            },
            body: JSON.stringify(payload),
        });

        console.log(`Results sent to webhook. Response status: ${response.status}`);

    }
});

            "Run Service Bot",
            "Run Bot",
            "Send Data to Service Bot Table",
            "Validate Total Data Count"
